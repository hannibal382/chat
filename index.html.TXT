<script src="https://cdn.streamelements.com/scripts/overlay.js"></script>

<script>
  const chat = document.getElementById("chat");

  window.addEventListener("onEventReceived", function (obj) {
    if (obj.detail.listener === "message") {
      const data = obj.detail.event.data;

      const msg = document.createElement("div");
      msg.classList.add("message");

      if (data.badges.broadcaster) msg.classList.add("broadcaster");
      if (data.badges.moderator) msg.classList.add("mod");
      if (data.badges.subscriber) msg.classList.add("subscriber");

      msg.innerHTML = `
        <span class="username">${data.displayName}</span>
        <span class="text">${data.text}</span>
      `;

      chat.appendChild(msg);

      setTimeout(() => {
        msg.remove();
      }, 15000);
    }
  });
</script>
